\section{Detailed Use Case Specifications}

\subsection{UC-01: Register Member}

\begin{tabular}{@{}p{3.5cm}p{9.5cm}@{}}
\toprule
\textbf{Field} & \textbf{Details} \\
\midrule
Use Case ID & UC-01 \\
Name & Register Member \\
Primary Actor & Visitor (prospective member) \\
Stakeholders & Admin, Organization \\
Trigger & Visitor accesses registration form \\
\bottomrule
\end{tabular}

\vspace{0.5em}
\textbf{Preconditions:}
\begin{itemize}
    \item Registration form is publicly accessible
    \item Visitor has valid NID and nominee information
\end{itemize}

\textbf{Main Flow:}
\begin{enumerate}
    \item Visitor opens the public registration form
    \item System displays form with required fields
    \item Visitor enters personal details (name, phone, email, NID, address)
    \item Visitor enters nominee details (name, NID, relationship)
    \item Visitor uploads required documents (photo, NID copy)
    \item Visitor accepts terms and submits form
    \item System validates input and stores registration
    \item System queues registration for admin review
    \item System displays confirmation message
\end{enumerate}

\textbf{Alternative Flows:}
\begin{itemize}
    \item \textbf{A1:} Validation fails $\rightarrow$ System highlights errors, visitor corrects and resubmits
    \item \textbf{A2:} Duplicate NID detected $\rightarrow$ System rejects with "already registered" message
\end{itemize}

\textbf{Postconditions:}
\begin{itemize}
    \item Registration record created with PENDING status
    \item Admin notification triggered
\end{itemize}

\subsection{UC-07: Apply for Loan}

\begin{tabular}{@{}p{3.5cm}p{9.5cm}@{}}
\toprule
\textbf{Field} & \textbf{Details} \\
\midrule
Use Case ID & UC-07 \\
Name & Apply for Loan \\
Primary Actor & Member \\
Stakeholders & Admin, Guarantor(s), Worker \\
Trigger & Member clicks "Apply for Loan" button \\
\bottomrule
\end{tabular}

\vspace{0.5em}
\textbf{Preconditions:}
\begin{itemize}
    \item Member account is active
    \item Minimum deposit requirement is satisfied (8,000 BDT)
    \item No existing active loan (unless policy allows multiple)
\end{itemize}

\textbf{Main Flow:}
\begin{enumerate}
    \item Member opens loan application form
    \item System displays maximum eligible amount based on caps
    \item Member enters requested amount and purpose
    \item System validates requested amount against eligibility
    \item If amount exceeds threshold, member selects guarantor(s)
    \item System stores loan request with PENDING\_GUARANTOR status
    \item System sends notification to selected guarantor(s)
\end{enumerate}

\textbf{Alternative Flows:}
\begin{itemize}
    \item \textbf{A1:} Requested amount exceeds allowed cap $\rightarrow$ System rejects with reason and displays max amount
    \item \textbf{A2:} Amount below threshold $\rightarrow$ No guarantor required, status set to PENDING\_APPROVAL
    \item \textbf{A3:} Selected guarantor is ineligible $\rightarrow$ System prompts to select different guarantor
\end{itemize}

\textbf{Postconditions:}
\begin{itemize}
    \item Loan request is either pending admin review (if complete) or awaiting guarantor/member action
    \item Audit log entry created
\end{itemize}

\subsection{UC-06: Raise Dispute}

\begin{tabular}{@{}p{3.5cm}p{9.5cm}@{}}
\toprule
\textbf{Field} & \textbf{Details} \\
\midrule
Use Case ID & UC-06 \\
Name & Raise Dispute \\
Primary Actor & Member \\
Stakeholders & Admin, Worker \\
Trigger & Member identifies incorrect amount on processed transaction \\
\bottomrule
\end{tabular}

\vspace{0.5em}
\textbf{Preconditions:}
\begin{itemize}
    \item Transaction is in PROCESSED status
    \item Transaction has not yet been confirmed or disputed
\end{itemize}

\textbf{Main Flow:}
\begin{enumerate}
    \item Member views transaction in dashboard
    \item Member clicks "Dispute" button
    \item System prompts for dispute reason and correct amount
    \item Member submits dispute
    \item System changes transaction status to DISPUTED
    \item System freezes transaction (prevents further state changes)
    \item System sends alert to Admin
\end{enumerate}

\textbf{Postconditions:}
\begin{itemize}
    \item Transaction status is DISPUTED
    \item Admin notification sent
    \item Audit log records dispute action
\end{itemize}

\subsection{UC-12: Default Settlement}

\begin{tabular}{@{}p{3.5cm}p{9.5cm}@{}}
\toprule
\textbf{Field} & \textbf{Details} \\
\midrule
Use Case ID & UC-12 \\
Name & Default Settlement \\
Primary Actor & Admin \\
Stakeholders & Member (borrower), Guarantor(s), Organization \\
Trigger & Admin initiates settlement for defaulted loan \\
\bottomrule
\end{tabular}

\vspace{0.5em}
\textbf{Preconditions:}
\begin{itemize}
    \item Loan is in DEFAULTED status
    \item Grace period has expired
\end{itemize}

\textbf{Main Flow:}
\begin{enumerate}
    \item Admin reviews defaulted loan details
    \item Admin initiates settlement waterfall
    \item System calculates outstanding balance
    \item System deducts from borrower's deposit account
    \item If insufficient, system deducts from guarantor(s) deposits proportionally
    \item If still insufficient, system flags as "Bad Debt" for charity fund write-off
    \item System updates loan status to SETTLED
    \item System sends notifications to affected parties
\end{enumerate}

\textbf{Postconditions:}
\begin{itemize}
    \item Loan marked as SETTLED
    \item All deductions recorded as transactions
    \item Affected accounts updated
    \item Complete audit trail recorded
\end{itemize}
