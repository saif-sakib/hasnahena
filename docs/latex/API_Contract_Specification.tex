\subsection{Base Configuration}

\begin{tabular}{@{}p{4cm}p{9cm}@{}}

\midrule

Character Encoding & UTF-8 \\
\bottomrule
\end{tabular}

\subsection{Authentication}
All protected endpoints require a Bearer token in the Authorization header:

\begin{lstlisting}
Authorization: Bearer <JWT_TOKEN>
\end{lstlisting}

Token type: JWT (JSON Web Token) compatible with session-based gateway policy.

\subsection{Response Envelope}
All API responses follow a consistent envelope structure:

\begin{lstlisting}
{
    "success": true | false,
    "data": { ... },
    "error": {
        "code": "ERROR_CODE",
        "message": "Human readable message"
    },
    "meta": {
        "timestamp": "ISO-8601",
        "requestId": "uuid"
    }
}
\end{lstlisting}

\section{Error Model}

\subsection{HTTP Status Codes}

\begin{tabular}{@{}p{1.5cm}p{3cm}p{8.5cm}@{}}
\toprule
\textbf{Code} & \textbf{Status} & \textbf{Description} \\
\midrule
200 & OK & Request successful \\
201 & Created & Resource created successfully \\
400 & Bad Request & Validation failure (malformed request) \\
401 & Unauthorized & Authentication required or failed \\
403 & Forbidden & Insufficient role permissions \\
404 & Not Found & Entity/resource not found \\
409 & Conflict & State conflict (e.g., already confirmed/disputed) \\
422 & Unprocessable Entity & Business rule violation (e.g., eligibility cap exceeded) \\
500 & Server Error & Internal server error \\
\bottomrule
\end{tabular}

\subsection{Error Response Example}

\begin{lstlisting}
{
    "success": false,
    "data": null,
    "error": {
        "code": "ELIGIBILITY_CAP_EXCEEDED",

            "requested": 100000,
            "maxAllowed": 50000
        }
    },
    "meta": {
        "timestamp": "2024-01-15T10:30:00Z",
        "requestId": "abc123-def456"
    }
}
\end{lstlisting}
